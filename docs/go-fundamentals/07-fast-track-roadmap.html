package main

import "fmt"

func main() {
    // 1. Variables
    var explicit int = 10
    inferred := "hello"
    const pi = 3.14
    _, _ = explicit, inferred
    _ = pi

    // 2. Collections
    nums := []int{1, 2, 3, 4, 5}
    nums = append(nums, 6)
    
    scores := map[string]int{
        "alice": 95,
        "bob":   87,
    }
    
    // 3. Control flow
    for i, v := range nums {
        if v > 3 {
            fmt.Printf("nums[%d] = %d\n", i, v)
        }
    }
    
    // 4. Struct
    type User struct {
        ID   int
        Name string
    }
    
    u := User{ID: 1, Name: "Alice"}
    fmt.Println(u)
    
    // 5. Function với multiple returns
    result, err := divide(10, 3)
    if err != nil {
        fmt.Println("Error:", err)
    } else {
        fmt.Println("Result:", result)
    }
    
    // 6. Pointer
    x := 42
    p := &x
    *p = 100
    fmt.Println("x =", x) // 100
}

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, fmt.Errorf("division by zero")
    }
    return a / b, nil
}

package main

import (
    "fmt"
    "time"
)

// Base fields
type AuditFields struct {
    CreatedAt time.Time
    UpdatedAt time.Time
}

// User với composition
type User struct {
    ID    int
    Name  string
    Email string
    AuditFields
}

// Constructor
func NewUser(name, email string) *User {
    now := time.Now()
    return &User{
        Name:  name,
        Email: email,
        AuditFields: AuditFields{
            CreatedAt: now,
            UpdatedAt: now,
        },
    }
}

// Methods
func (u *User) UpdateEmail(email string) {
    u.Email = email
    u.UpdatedAt = time.Now()
}

func (u User) String() string {
    return fmt.Sprintf("User{id=%d, name=%s, email=%s}", u.ID, u.Name, u.Email)
}

// interface pattern
type UserRepository interface {
    FindByID(id int) (*User, error)
    Create(user *User) error
}

// In-memory implementation
type InMemoryUserRepo struct {
    users map[int]*User
    nextID int
}

func (r *InMemoryUserRepo) FindByID(id int) (*User, error) {
    user, ok := r.users[id]
    if !ok {
        return nil, fmt.Errorf("user %d: %w", id, ErrNotFound)
    }
    return user, nil
}

// Sentinel error
var ErrNotFound = errors.New("not found")

// Custom error type
type ValidationError struct {
    Field   string
    Message string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("validation '%s': %s", e.Field, e.Message)
}

// main.go
package main

import (
    "encoding/json"
    "log"
    "net/http"
    "strconv"
    "strings"
)

type User struct {
    ID    int    `json:"id"`
    Name  string `json:"name"`
    Email string `json:"email"`
}

var users = map[int]*User{}
var nextID = 1

func main() {
    mux := http.NewServeMux()
    
    mux.HandleFunc("GET /health", healthHandler)
    mux.HandleFunc("GET /api/users", listUsersHandler)
    mux.HandleFunc("POST /api/users", createUserHandler)
    mux.HandleFunc("GET /api/users/{id}", getUserHandler)
    
    log.Println("Server on :8080")
    log.Fatal(http.ListenAndServe(":8080", mux))
}

func healthHandler(w http.ResponseWriter, r *http.Request) {
    json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
}

func listUsersHandler(w http.ResponseWriter, r *http.Request) {
    list := make([]*User, 0, len(users))
    for _, u := range users {
        list = append(list, u)
    }
    json.NewEncoder(w).Encode(list)
}

func createUserHandler(w http.ResponseWriter, r *http.Request) {
    var req struct {
        Name  string `json:"name"`
        Email string `json:"email"`
    }
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "invalid json", http.StatusBadRequest)
        return
    }
    
    user := &User{ID: nextID, Name: req.Name, Email: req.Email}
    users[nextID] = user
    nextID++
    
    w.WriteHeader(http.StatusCreated)
    json.NewEncoder(w).Encode(user)
}

func getUserHandler(w http.ResponseWriter, r *http.Request) {
    id, err := strconv.Atoi(r.PathValue("id"))
    if err != nil {
        http.Error(w, "invalid id", http.StatusBadRequest)
        return
    }
    
    user, ok := users[id]
    if !ok {
        http.Error(w, "not found", http.StatusNotFound)
        return
    }
    
    json.NewEncoder(w).Encode(user)
}

func TestDivide(t *testing.T) {
    tests := []struct {
        name    string
        a, b    float64
        want    float64
        wantErr bool
    }{
        {"positive", 10, 3, 3.333, false},
        {"zero divisor", 10, 0, 0, true},
        {"negative", -10, 3, -3.333, false},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got, err := divide(tt.a, tt.b)
            if (err != nil) != tt.wantErr {
                t.Errorf("divide() error = %v, wantErr %v", err, tt.wantErr)
                return
            }
            if !tt.wantErr && abs(got-tt.want) > 0.001 {
                t.Errorf("divide() = %v, want %v", got, tt.want)
            }
        })
    }
}

func abs(x float64) float64 {
    if x < 0 {
        return -x
    }
    return x
}

package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    var wg sync.WaitGroup
    
    for i := 0; i < 5; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            fmt.Printf("Worker %d starting\n", id)
            time.Sleep(time.Second)
            fmt.Printf("Worker %d done\n", id)
        }(i)
    }
    
    wg.Wait()
    fmt.Println("All workers done!")
}

// Worker pool
func workerPool(jobs <-chan int, results chan<- int, workers int) {
    var wg sync.WaitGroup
    
    for i := 0; i < workers; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            for job := range jobs {
                fmt.Printf("Worker %d processing job %d\n", id, job)
                time.Sleep(100 * time.Millisecond)
                results <- job * 2
            }
        }(i)
    }
    
    wg.Wait()
    close(results)
}

// Context timeout
func fetchWithTimeout(ctx context.Context, url string) ([]byte, error) {
    ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()
    
    req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
    if err != nil {
        return nil, err
    }
    
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    return io.ReadAll(resp.Body)
}

// cmd/api/main.go
func main() {
    // Dependencies
    userRepo := repository.NewInMemoryUserRepository()
    userService := service.NewUserService(userRepo)
    userHandler := handler.NewUserHandler(userService)
    
    // Router
    mux := http.NewServeMux()
    mux.HandleFunc("GET /api/users", userHandler.ListUsers)
    mux.HandleFunc("POST /api/users", userHandler.CreateUser)
    mux.HandleFunc("GET /api/users/{id}", userHandler.GetUser)
    
    // Server
    server := &http.Server{
        Addr:    ":8080",
        Handler: middleware.Recovery(middleware.Logging(mux)),
    }
    
    // Graceful shutdown
    go func() {
        log.Println("Server starting on :8080")
        if err := server.ListenAndServe(); err != http.ErrServerClosed {
            log.Fatal(err)
        }
    }()
    
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit
    
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    server.Shutdown(ctx)
}

Todo API:
- POST /todos         → Create todo
- GET /todos          → List todos
- GET /todos/{id}     → Get todo
- PUT /todos/{id}     → Update todo
- DELETE /todos/{id}  → Delete todo

Học được: CRUD, JSON, routing, testing

URL Shortener:
- POST /shorten       → Tạo short URL
- GET /{shortID}      → Redirect to original

Học được: map/storage, HTTP redirect, ID generation

Web Crawler:
- Input: list of URLs
- Process: fetch concurrently với worker pool
- Output: collected results

Học được: goroutine, channel, worker pool, context cancellation

API Client:
- Fetch nhiều URLs concurrently
- Rate limit: max N requests/second
- Retry with exponential backoff
- Timeout per request

Học được: ticker, semaphore, retry pattern, error handling
